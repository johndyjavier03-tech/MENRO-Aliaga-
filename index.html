""" Budget Log - weekly collections and expenses, stored in a local SQLite file. Run: python app.py Open: http://localhost:8000 No packages to install (Python 3.8+ only). Data is saved in budget_log.db next to this file. """ import html import sqlite3 from datetime import date, datetime, timedelta from http.server import BaseHTTPRequestHandler, HTTPServer from pathlib import Path from urllib.parse import parse_qs DB_PATH = Path(__file__).with_name("budget_log.db") HOST, PORT = "localhost", 8000 CURRENCY = "₱" # ---------- database ---------- def db(): conn = sqlite3.connect(DB_PATH) conn.row_factory = sqlite3.Row return conn def init_db(): with db() as c: c.executescript( """ CREATE TABLE IF NOT EXISTS collections ( id INTEGER PRIMARY KEY AUTOINCREMENT, week_start TEXT NOT NULL, amount REAL NOT NULL CHECK (amount > 0), collected_by TEXT, notes TEXT ); CREATE TABLE IF NOT EXISTS expenses ( id INTEGER PRIMARY KEY AUTOINCREMENT, expense_date TEXT NOT NULL, description TEXT NOT NULL, category TEXT, amount REAL NOT NULL CHECK (amount > 0) ); """ ) # ---------- calculations ---------- def monday_of(d: date) -> date: return d - timedelta(days=d.weekday()) def get_summary(): with db() as c: collected = c.execute("SELECT COALESCE(SUM(amount),0) FROM collections").fetchone()[0] spent = c.execute("SELECT COALESCE(SUM(amount),0) FROM expenses").fetchone()[0] return {"collected": collected, "expenses": spent, "remaining": collected - spent} def get_category_totals(): with db() as c: return c.execute( """SELECT COALESCE(NULLIF(category,''),'Uncategorized') AS category, SUM(amount) AS total FROM expenses GROUP BY 1 ORDER BY total DESC""" ).fetchall() def add_collection(week, amount, by, notes): week_start = monday_of(datetime.strptime(week, "%Y-%m-%d").date()).isoformat() with db() as c: c.execute( "INSERT INTO collections (week_start, amount, collected_by, notes) VALUES (?,?,?,?)", (week_start, amount, by, notes), ) def add_expense(when, description, category, amount): datetime.strptime(when, "%Y-%m-%d") # validates the date with db() as c: c.execute( "INSERT INTO expenses (expense_date, description, category, amount) VALUES (?,?,?,?)", (when, description, category, amount), ) def delete_row(table, row_id): if table not in ("collections", "expenses"): return with db() as c: c.execute(f"DELETE FROM {table} WHERE id = ?", (row_id,)) # ---------- page ---------- def money(n): return f"{CURRENCY}{n:,.2f}" def esc(s): return html.escape(s or "") CSS = """ :root{ --bg:#eef1ee; --panel:#fbfcfb; --ink:#1b2a2f; --muted:#5c6b70; --line:#d3dbd6; --accent:#0e6e6d; --accent-ink:#ffffff; --bad:#a73a24; --bad-bg:#f8e6e1; } @media (prefers-color-scheme: dark){ :root{ --bg:#121a1c; --panel:#192326; --ink:#e4ece8; --muted:#93a3a0; --line:#2b3a3d; --accent:#46b3ad; --accent-ink:#0b1415; --bad:#f0907a; --bad-bg:#3a1f1a; } } *{box-sizing:border-box} body{margin:0;background:var(--bg);color:var(--ink); font:16px/1.5 "Segoe UI",system-ui,-apple-system,Roboto,sans-serif} main{max-width:1040px;margin:0 auto;padding:32px 20px 64px} h1{font-size:1.5rem;margin:0 0 4px} h2{font-size:1.1rem;margin:0 0 12px} .sub{color:var(--muted);margin:0 0 24px} .num{font-variant-numeric:tabular-nums} /* The remaining budget is the one thing that gets visual weight. */ .remaining{background:var(--accent);color:var(--accent-ink);border-radius:10px; padding:24px 28px;margin-bottom:16px} .remaining.over{background:var(--bad);color:#fff} .remaining .label{opacity:.85} .remaining .big{font-size:clamp(2.2rem,6vw,3.4rem);font-weight:700;line-height:1.1} .remaining .eq{margin-top:10px;opacity:.9} .warn{background:var(--bad-bg);color:var(--bad);border-left:4px solid var(--bad); padding:10px 14px;border-radius:4px;margin-bottom:16px} .grid{display:grid;grid-template-columns:1fr 1fr;gap:20px;margin-top:24px} @media (max-width:760px){.grid{grid-template-columns:1fr}} section{background:var(--panel);border:1px solid var(--line);border-radius:10px;padding:20px} form{display:grid;grid-template-columns:1fr 1fr;gap:10px;margin-bottom:16px} form .full{grid-column:1/-1} label{display:block;font-size:.85rem;color:var(--muted);margin-bottom:2px} input{width:100%;padding:8px 10px;border:1px solid var(--line);border-radius:6px; background:var(--bg);color:var(--ink);font:inherit} input:focus-visible,button:focus-visible{outline:2px solid var(--accent);outline-offset:2px} button{font:inherit;cursor:pointer;border:0;border-radius:6px;padding:9px 14px} .primary{background:var(--accent);color:var(--accent-ink);font-weight:600} .link{background:none;color:var(--muted);padding:2px 6px;text-decoration:underline} .link:hover{color:var(--bad)} table{width:100%;border-collapse:collapse;font-size:.95rem} th{text-align:left;font-weight:600;color:var(--muted);border-bottom:1px solid var(--line);padding:6px 4px} td{padding:7px 4px;border-bottom:1px solid var(--line);vertical-align:top} td.r,th.r{text-align:right} .empty{color:var(--muted);padding:8px 0} .small{color:var(--muted);font-size:.85rem} .cats{margin-top:24px} """ def render_page(error=""): s = get_summary() with db() as c: cols = c.execute("SELECT * FROM collections ORDER BY week_start DESC, id DESC").fetchall() exps = c.execute("SELECT * FROM expenses ORDER BY expense_date DESC, id DESC").fetchall() cats = get_category_totals() over = s["remaining"] < 0 today = date.today().isoformat() col_rows = "".join( f"""{esc(r['week_start'])} {esc(r['collected_by'])}
{esc(r['notes'])}
{money(r['amount'])}
""" for r in cols ) or 'No collections yet. Add the first one above.' exp_rows = "".join( f"""{esc(r['expense_date'])} {esc(r['description'])}
{esc(r['category'])}
{money(r['amount'])}
""" for r in exps ) or 'No expenses yet. Add the first one above.' cat_rows = "".join( f"{esc(r['category'])}{money(r['total'])}" for r in cats ) banner = f'
{esc(error)}
' if error else "" over_note = ( '
Expenses are higher than the money collected.
' if over else "" ) return f""" Budget Log

Budget Log

Weekly collections and expenses, saved on this computer.

{banner}
Remaining budget
{money(s['remaining'])}
{money(s['collected'])} collected − {money(s['expenses'])} spent
{over_note}

Money collected

{col_rows}
Week of (Mon)DetailsAmount

Expenses

{exp_rows}
DateDetailsAmount

Spending by category

{cat_rows or ''}
Nothing spent yet.
""" # ---------- server ---------- class Handler(BaseHTTPRequestHandler): def _send(self, body, status=200, location=None): data = body.encode("utf-8") self.send_response(status) if location: self.send_header("Location", location) self.send_header("Content-Type", "text/html; charset=utf-8") self.send_header("Content-Length", str(len(data))) self.end_headers() self.wfile.write(data) def do_GET(self): if self.path.split("?")[0] == "/": self._send(render_page()) else: self._send("Not found", 404) def do_POST(self): length = int(self.headers.get("Content-Length", 0)) form = {k: v[0].strip() for k, v in parse_qs(self.rfile.read(length).decode()).items()} try: if self.path == "/collection": add_collection(form["week"], _amount(form["amount"]), form.get("by", ""), form.get("notes", "")) elif self.path == "/expense": add_expense(form["date"], form["description"], form.get("category", ""), _amount(form["amount"])) elif self.path == "/delete": delete_row(form["table"], int(form["id"])) else: return self._send("Not found", 404) except (KeyError, ValueError): return self._send(render_page("That entry could not be saved. " "Check the date and enter an amount above zero."), 400) self._send("", 303, location="/") def log_message(self, fmt, *args): # keep the console quiet pass def _amount(text): value = round(float(text), 2) if value <= 0: raise ValueError("amount must be positive") return value if __name__ == "__main__": init_db() print(f"Budget Log running at http://{HOST}:{PORT} (Ctrl+C to stop)") HTTPServer((HOST, PORT), Handler).serve_forever()
