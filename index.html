"""Budget log: weekly collections + expenses, stored in a single SQLite file (budget.db).""" import sqlite3 from datetime import date from pathlib import Path from flask import Flask, g, jsonify, request, send_from_directory DB = Path(__file__).with_name("budget.db") app = Flask(__name__, static_folder="static") SCHEMA = """ CREATE TABLE IF NOT EXISTS entries ( id INTEGER PRIMARY KEY AUTOINCREMENT, kind TEXT NOT NULL CHECK (kind IN ('collection', 'expense')), day TEXT NOT NULL, -- YYYY-MM-DD amount REAL NOT NULL CHECK (amount > 0), note TEXT NOT NULL DEFAULT '' ); """ def db(): if "db" not in g: g.db = sqlite3.connect(DB) g.db.row_factory = sqlite3.Row g.db.execute(SCHEMA) return g.db @app.teardown_appcontext def close_db(_): conn = g.pop("db", None) if conn: conn.close() def summary(): row = db().execute(""" SELECT COALESCE(SUM(CASE WHEN kind='collection' THEN amount END), 0) AS collected, COALESCE(SUM(CASE WHEN kind='expense' THEN amount END), 0) AS spent FROM entries""").fetchone() return {"collected": row["collected"], "spent": row["spent"], "remaining": round(row["collected"] - row["spent"], 2)} @app.get("/") def index(): return send_from_directory("static", "index.html") @app.get("/api/entries") def list_entries(): rows = db().execute("SELECT * FROM entries ORDER BY day DESC, id DESC").fetchall() return jsonify(entries=[dict(r) for r in rows], summary=summary()) @app.post("/api/entries") def add_entry(): d = request.get_json(force=True) kind, note = d.get("kind"), (d.get("note") or "").strip() try: amount = round(float(d.get("amount")), 2) day = date.fromisoformat(d.get("day") or date.today().isoformat()).isoformat() except (TypeError, ValueError): return jsonify(error="Enter a valid amount and date."), 400 if kind not in ("collection", "expense") or amount <= 0: return jsonify(error="Amount must be greater than zero."), 400 db().execute("INSERT INTO entries (kind, day, amount, note) VALUES (?,?,?,?)", (kind, day, amount, note)) db().commit() return list_entries(), 201 @app.delete("/api/entries/") def delete_entry(entry_id): db().execute("DELETE FROM entries WHERE id = ?", (entry_id,)) db().commit() return list_entries() if __name__ == "__main__": app.run(debug=True)
